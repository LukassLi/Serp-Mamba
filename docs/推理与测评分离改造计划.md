# 推理与测评分离改造计划

## 一、现状问题

当前 [test.py](test.py) 将**推理**（模型前向传播、保存预测掩码）和**测评**（计算指标、写入 output.txt）耦合在同一个文件中：

```
test.py（~427 行）
├── 模型构建 + 配置加载（1-163 行）
├── 指标函数：calculate_bm, calculate_se_sp_pr, calculate_auc, calculate_hd95, calculate_mcc, calculate_metric_percase1/2（166-232 行）
├── test_single_volume_fast()：推理 + 测评混合（235-293 行）
└── Inference()：循环推理 + 聚合指标 + 写文件（296-426 行）
```

**痛点**：
1. 不同数据集需要不同评价指标（如 FIVES 需要 AUC/HD95，DRIVE 不需要），但当前所有指标硬编码在一起
2. 新增数据集的指标需求会持续膨胀 test.py
3. 指标函数分散在三处（test.py、utils/metrics.py、val_2D.py），大量重复代码
4. 无法单独对已保存的预测结果重新测评（必须重新跑推理）

## 二、目标架构

```
Serp_Mamba/
├── test.py                      # 纯推理：加载模型 → 前向传播 → 保存预测掩码 + 概率图
├── evaluate.py                  # 测评入口：加载配置 → 找到预测结果 → 调用对应 evaluator
├── utils/
│   └── metrics.py               # 公共指标工具（所有指标函数集中于此，供所有模块复用）
├── evaluators/
│   ├── __init__.py              # 注册表：数据集名 → evaluator 类
│   ├── base.py                  # BaseEvaluator 基类（定义统一接口）
│   └── default.py               # DefaultEvaluator（Dice, IoU, MCC, BM — 所有数据集通用）
├── configs/
│   ├── drive.yaml               # 增加 evaluator: "default"
│   ├── fives.yaml               # 增加 evaluator: "fives"
│   └── ...
└── val_2D.py                    # 训练验证：改为调用 utils/metrics.py 中的公共函数
```

## 三、详细设计

### 3.1 公共指标工具 — `utils/metrics.py`（改造现有文件）

将分散在 test.py、val_2D.py 中的指标函数**统一提取**到此处，消除重复。

```python
# utils/metrics.py — 改造后

import numpy as np
from medpy import metric as medpy_metric

# ── 通用单值指标 ──

def calc_dice(pred, gt):
    """Dice 系数 (F1)"""
    if pred.sum() == 0:
        return 0.0
    return medpy_metric.binary.dc(pred, gt)

def calc_iou(pred, gt):
    """IoU / Jaccard 指数"""
    if pred.sum() == 0:
        return 0.0
    return medpy_metric.binary.jc(pred, gt)

def calc_hd95(pred, gt):
    """Hausdorff Distance 95th percentile"""
    if pred.sum() > 0 and gt.sum() > 0:
        return medpy_metric.binary.hd95(pred, gt)
    return 0.0

def calc_asd(pred, gt):
    """Average Surface Distance"""
    if pred.sum() > 0 and gt.sum() > 0:
        return medpy_metric.binary.asd(pred, gt)
    return 0.0

def calc_auc(pred_prob, gt):
    """AUC (需要 softmax 概率，不是二值预测)"""
    from sklearn.metrics import roc_auc_score
    gt_flat = gt.flatten().astype(int)
    prob_flat = pred_prob.flatten()
    if len(np.unique(gt_flat)) < 2:
        return 0.0
    return roc_auc_score(gt_flat, prob_flat)

# ── 基于混淆矩阵的指标族 ──

def confusion_matrix(pred, gt):
    """返回 (TP, FP, TN, FN)"""
    TP = ((pred == 1) & (gt == 1)).sum()
    FP = ((pred == 1) & (gt == 0)).sum()
    TN = ((pred == 0) & (gt == 0)).sum()
    FN = ((pred == 0) & (gt == 1)).sum()
    return TP, FP, TN, FN

def calc_sensitivity(pred, gt):
    """Sensitivity / Recall / TPR — 血管检出率"""
    TP, FP, TN, FN = confusion_matrix(pred, gt)
    return TP / (TP + FN) if (TP + FN) > 0 else 0.0

def calc_specificity(pred, gt):
    """Specificity / TNR — 背景准确率"""
    TP, FP, TN, FN = confusion_matrix(pred, gt)
    return TN / (TN + FP) if (TN + FP) > 0 else 0.0

def calc_precision(pred, gt):
    """Precision — 预测血管精确率"""
    TP, FP, TN, FN = confusion_matrix(pred, gt)
    return TP / (TP + FP) if (TP + FP) > 0 else 0.0

def calc_mcc(pred, gt):
    """Matthews Correlation Coefficient"""
    TP, FP, TN, FN = confusion_matrix(pred, gt)
    denom = np.sqrt(float(TP+FP) * float(TP+FN) * float(TN+FP) * float(TN+FN))
    return ((TP * TN) - (FP * FN)) / denom if denom > 0 else 0.0

def calc_bm(pred, gt):
    """Bookmaker Informedness (Youden's J) = TPR + TNR - 1"""
    return calc_sensitivity(pred, gt) + calc_specificity(pred, gt) - 1

# ── 批量计算 ──

def compute_metrics(pred, gt, metric_names, pred_prob=None):
    """
    统一接口：根据指标名列表批量计算。

    Args:
        pred: 二值预测 (numpy array, 0/1)
        gt: 二值标签 (numpy array, 0/1)
        metric_names: 指标名列表，如 ["dice", "iou", "se", "auc"]
        pred_prob: softmax 概率图 (仅 AUC 需要)

    Returns:
        dict: {metric_name: float_value}
    """
    results = {}
    for name in metric_names:
        fn = METRIC_REGISTRY.get(name)
        if fn is None:
            raise ValueError(f"未知指标: {name}")
        if name == "auc":
            results[name] = fn(pred_prob, gt)
        else:
            results[name] = fn(pred, gt)
    return results

# 注册表：指标名 → 计算函数
METRIC_REGISTRY = {
    "dice": calc_dice,
    "iou": calc_iou,
    "mcc": calc_mcc,
    "bm": calc_bm,
    "se": calc_sensitivity,
    "sp": calc_specificity,
    "pr": calc_precision,
    "auc": calc_auc,       # 特殊：需要 pred_prob
    "hd95": calc_hd95,
    "asd": calc_asd,
}
```

**复用关系**：
- `val_2D.py` 中 `calculate_metric_percase1` / `calculate_metric_percase2` → 改为调用 `calc_dice` / `calc_iou`
- `test.py` 中的 `calculate_bm` / `calculate_se_sp_pr` / `calculate_auc` / `calculate_hd95` / `calculate_mcc` → 全部删除，改用 `utils/metrics.py`
- `utils/metrics.py` 原有的 `cal_dice` / `calculate_metric_percase` / `dice` → 整合为上述统一接口

def run_inference(net, image, patch_size):
    """纯推理：图像 → softmax 概率 + 二值预测"""
    image_np = image.squeeze().cpu().detach().numpy()
    x, y = image_np.shape
    zoomed = zoom(image_np, (patch_size[0] / x, patch_size[1] / y), order=0)
    input_tensor = torch.from_numpy(zoomed).unsqueeze(0).unsqueeze(0).float().cuda()

        softmax_probs = torch.softmax(net(input_tensor), dim=1)
        pred = torch.argmax(softmax_probs, dim=1).cpu().detach().numpy().squeeze()

    prediction = zoom(pred, (x / patch_size[0], y / patch_size[1]), order=0)
    prob_map = zoom(softmax_probs[0, 1].cpu().detach().numpy(),
                    (x / patch_size[0], y / patch_size[1]), order=0)

    return prediction, prob_map

### 3.3 Evaluator 框架 — `evaluators/`

#### `evaluators/base.py` — 基类

```python
from abc import ABC, abstractmethod
from utils.metrics import compute_metrics

class BaseEvaluator(ABC):
    """评测器基类，定义统一接口"""

    @abstractmethod
    def get_metric_names(self):
        """返回本评测器计算的指标名列表"""
        pass

    def evaluate_single(self, prediction, label, pred_prob=None):
        """
        评估单张图片，默认实现直接调用 compute_metrics。
        子类可覆盖以自定义计算逻辑。
        """
        return compute_metrics(prediction, label, self.get_metric_names(), pred_prob=pred_prob)

    def aggregate(self, per_image_results):
        """
        汇总逐图结果 → 均值 ± 标准差。
        返回 dict: {metric_name: (mean, std)}
        """
        metric_names = self.get_metric_names()
        perf = {}
        for name in metric_names:
            values = [r[name] for r in per_image_results]
            perf[name] = (np.mean(values), np.std(values))
        return perf

    def format_per_image(self, case_name, metrics):
        """格式化单图结果为字符串"""
        parts = [f"{k}={v:.6f}" for k, v in metrics.items()]
        return f"  {case_name}: " + ", ".join(parts) + "\n"

    def format_summary(self, perf, model_name, split, num_samples):
        """格式化汇总结果为字符串"""
        lines = [
            f"model_name = {model_name}\n",
            f"split = {split}\n",
            f"num_samples = {num_samples}\n",
        ]
        for name, (mean_val, std_val) in perf.items():
            lines.append(f"{name} = mean-sd = {mean_val}-{std_val}\n")
        return "".join(lines)
```

from evaluators.default import DefaultEvaluator
from evaluators.fives import FIVESEvaluator

EVALUATOR_REGISTRY = {
    "default": DefaultEvaluator,
    "fives": FIVESEvaluator,
    # "drive": DefaultEvaluator,    # 使用默认即可
    # "fundus": DefaultEvaluator,   # 使用默认即可
    # "prime_fp20": DefaultEvaluator,
}

def get_evaluator(config):
    """
    根据配置获取评测器实例。
    优先读 config["evaluator"]，否则回退到 "default"。
    """
    evaluator_name = config.get("evaluator", "default")
    cls = EVALUATOR_REGISTRY.get(evaluator_name)
    if cls is None:
        raise ValueError(f"未知评测器: {evaluator_name}，可选: {list(EVALUATOR_REGISTRY.keys())}")
    return cls()
